  f.split("/").pop().replace(/(\.test)?\.ts$/, "");
const exporters = (await repo.files("src/export/*.ts"))
  .map(name).filter((n) => n !== "index" && n !== "base");
const tested = new Set(
  (await repo.files("test/export/*.test.ts")).map(name));
const registry = repo.read("src/export/index.ts").text;
exporters.map((n) => ({ exporter: n, tested: tested.has(n),
  registered: registry.includes(n) }));
